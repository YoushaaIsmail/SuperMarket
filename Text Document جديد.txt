' ==================== ENTITIES ====================

class User {
    + primary_key(id) : int
    + username : varchar(150)
    + email : varchar(254)
    + first_name : varchar(150)
    + last_name : varchar(150)
    + is_staff : boolean
    + is_active : boolean
    + date_joined : datetime
    + role : varchar(20)  [manager, accountant]
    + phone : varchar(20)
    + address : text
}

class Category {
    + primary_key(id) : int
    + name : varchar(100) [unique]
    + description : text
}

class Unite {
    + primary_key(id) : int
    + name : varchar(100) [unique]
}

class Supplier {
    + primary_key(id) : int
    + name : varchar(200)
    + phone : varchar(20)
    + email : varchar(254)
    + address : text
}

class Customer {
    + primary_key(id) : int
    + name : varchar(200)
    + phone : varchar(20)
    + email : varchar(254)
    + address : text
}

class Product {
    + primary_key(id) : int
    + name : varchar(200)
    + barcode : varchar(100) [unique]
    + sale_price : decimal(10,2)
    + sale_priceAmount : decimal(10,2)
    + Pakage : int
    + CountAlarm: int
    + created_at : datetime
    + updated_at : datetime
    -- foreign keys --
    + category_id : int
    + unite_id : int
}

class BatchProduct {
    + primary_key(id) : int
    + batch_number : varchar(100)
    + quantity : int
    + expiration_date : date
    + purchase_price : decimal(10,2)
    + created_at : datetime
    -- foreign key --
    + product_id : int
}

class ExchangeRate {
    + primary_key(id) : int
    + date : date
    + rate : decimal(12,2)
}

class PurchaseInvoice {
    + primary_key(id) : int
    + invoice_number : varchar(50) [unique]
    + date : date
    + is_paid : boolean
    + notes : text
    -- foreign keys --
    + supplier_id : int
    + exchange_rate_id : int
}

class PurchaseInvoiceDetail {
    + primary_key(id) : int
    + quantity : int
    + Is_package : boolean
    + price : decimal(10,2)
    -- foreign keys --
    + invoice_id : int
    + product_id : int
}

class SalesInvoice {
    + primary_key(id) : int
    + invoice_number : varchar(50) [unique]
    + date : date
    + is_paid : boolean
    + notes : text
    -- foreign keys --
    + customer_id : int
    + exchange_rate_id : int
}

class SalesInvoiceDetail {
    + primary_key(id) : int
    + quantity : int
    + price : decimal(10,2)
    + Is_package : boolean
    -- foreign keys --
    + invoice_id : int
    + product_id : int
}

class PaymentSales {
    + primary_key(id) : int
    + invoice_type : char(1)  [P/S]
    + amount : decimal(12,2)
    + date : date
    + method : varchar(20)
    + reference : varchar(100)
    + notes : text
    -- foreign keys --
    + invoice_id : int  (FK to SalesInvoice)
    + exchange_rate_used_id : int
}

class PaymentPurchase {
    + primary_key(id) : int
    + invoice_type : char(1)  [P/S]
    + amount : decimal(12,2)
    + date : date
    + method : varchar(20)
    + reference : varchar(100)
    + notes : text
    -- foreign keys --
    + invoice_id : int  (FK to PurchaseInvoice)
    + exchange_rate_used_id : int
}

class Employee {
    + primary_key(id) : int
    + name : varchar(200)
    + position : varchar(100)
    + default_base_salary : decimal(10,2)
    + hire_date : date
    + phone : varchar(20)
    + email : varchar(254)
    + bank_account : varchar(100)
    + is_active : boolean
    + created_at : datetime
    -- foreign key (one-to-one) --
    + user_id : int  (FK to User)
}

class ActionMoney_FullAll {
    + primary_key(id) : int
    + relationship : varchar(20)  [Allowance, Penalty]
    + description : varchar(200)
    + amount : decimal(10,2)
    + created_at : datetime
    -- foreign key --
    + Employee_id : int
}

class SalaryPayment {
    + primary_key(id) : int
    + payment_date : date
    + amount : decimal(10,2)
    + payment_method : varchar(50)
    + notes : text
    + created_at : datetime
    -- foreign key --
    + Employee_id : int
}

class Action_Money_Month {
    + primary_key(id) : int
    + relationship : varchar(20)  [Bonus, advance, Deduction]
    + description : varchar(200)
    + amount : decimal(10,2)
    + created_at : datetime
    -- foreign key --
    + Employee_id : int
}

class CostCategory {
    + primary_key(id) : int
    + name : varchar(100) [unique]
    + description : text
}

class Cost {
    + primary_key(id) : int
    + amount : decimal(10,2)
    + date : date
    + description : text
    -- foreign key --
    + category_id : int
}

' ==================== RELATIONSHIPS ====================

' Product relationships
Product "1" -- "0..*" Category : "belongs to"
Product "1" -- "0..*" Unite : "measured in"
Product "1" -- "0..*" BatchProduct : "has batches"

' BatchProduct -> Product (already shown)

' PurchaseInvoice relationships
PurchaseInvoice "1" -- "0..*" Supplier : "from"
PurchaseInvoice "1" -- "0..*" ExchangeRate : "uses rate"
PurchaseInvoice "1" -- "0..*" PurchaseInvoiceDetail : "contains"

' PurchaseInvoiceDetail relationships
PurchaseInvoiceDetail "1" -- "0..*" Product : "purchases"

' SalesInvoice relationships
SalesInvoice "1" -- "0..*" Customer : "to"
SalesInvoice "1" -- "0..*" ExchangeRate : "uses rate"
SalesInvoice "1" -- "0..*" SalesInvoiceDetail : "contains"

' SalesInvoiceDetail relationships
SalesInvoiceDetail "1" -- "0..*" Product : "sells"

' PaymentSales relationships
PaymentSales "1" -- "0..*" SalesInvoice : "for"
PaymentSales "1" -- "0..*" ExchangeRate : "uses rate"

' PaymentPurchase relationships
PaymentPurchase "1" -- "0..*" PurchaseInvoice : "for"
PaymentPurchase "1" -- "0..*" ExchangeRate : "uses rate"

' Employee relationships
Employee "1" -- "1" User : "linked to"
Employee "1" -- "0..*" ActionMoney_FullAll : "has full allowances"
Employee "1" -- "0..*" SalaryPayment : "receives"
Employee "1" -- "0..*" Action_Money_Month : "has monthly adjustments"

' Cost relationships
Cost "1" -- "0..*" CostCategory : "belongs to"

@enduml
